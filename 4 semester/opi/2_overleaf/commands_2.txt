function Apply-Zip {
    param(
        [Parameter(Mandatory=$true)][string]$Rev
    )

    $num = $Rev -replace '^r', ''
    $zipPath = "$zips\commit$num.zip"

    if (!(Test-Path $zipPath)) {
        throw "Archive not found: $zipPath"
    }

    $tmpRoot = "$root\tmp"

    if (!(Test-Path $tmpRoot)) {
        New-Item -ItemType Directory -Path $tmpRoot | Out-Null
    }

    $tempDir = "$tmpRoot\temp_extract_$Rev"

    if (Test-Path $tempDir) {
        Remove-Item -Recurse -Force $tempDir
    }

    New-Item -ItemType Directory -Path $tempDir | Out-Null

    # Вместо Expand-Archive используем tar.exe
    & tar.exe -xf $zipPath -C $tempDir

    if ($LASTEXITCODE -ne 0) {
        throw "Failed to extract archive: $zipPath"
    }

    # Удаляем всё из рабочей копии, кроме .git
    Get-ChildItem -Force | Where-Object {
        $_.Name -ne ".git"
    } | Remove-Item -Recurse -Force

    # Если внутри архива одна корневая папка, берём её содержимое
    $items = @(Get-ChildItem $tempDir -Force)

    if ($items.Count -eq 1 -and $items[0].PSIsContainer) {
        $sourceDir = $items[0].FullName
    } else {
        $sourceDir = $tempDir
    }

    Get-ChildItem $sourceDir -Force | Where-Object {
        $_.Name -ne ".git"
    } | ForEach-Object {
        Copy-Item $_.FullName -Destination (Get-Location).Path -Recurse -Force
    }

    Remove-Item -Recurse -Force $tempDir
}

function Commit-ZipRevision {
    param(
        [Parameter(Mandatory=$true)][string]$Rev,
        [Parameter(Mandatory=$true)][string]$Actor,
        [Parameter(Mandatory=$true)][string]$Branch
    )

    Write-Host "Creating $Rev by $Actor on $Branch"

    Apply-Zip $Rev
    git add -A
    git commit --allow-empty -m "${Rev}: ${Actor} change on ${Branch}"
    git tag $Rev
}

function Merge-ZipRevision {
    param(
        [Parameter(Mandatory=$true)][string]$Rev,
        [Parameter(Mandatory=$true)][string]$Source,
        [Parameter(Mandatory=$true)][string]$Message
    )

    Write-Host "Merging $Source as $Rev"

    git merge --no-ff --no-commit -s ours $Source

    if ($LASTEXITCODE -ne 0) {
        throw "Git merge failed for $Rev from source $Source"
    }

    Apply-Zip $Rev
    git add -A
    git commit --allow-empty -m $Message
    git tag $Rev
}

function Push-Current {
    param(
        [Parameter(ValueFromRemainingArguments=$true)]
        [string[]]$Tags
    )

    git push -u origin HEAD

    foreach ($t in $Tags) {
        git push origin $t
    }
}
